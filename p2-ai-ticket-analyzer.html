Which complaint categories have the highest severity and lowest satisfaction ratings?

SELECT category, COUNT(*) AS ticket_count, ROUND(100.0 * SUM(CASE WHEN severity = 'High' THEN 1 ELSE 0 END) / COUNT(*), 1) AS high_severity_pct, ROUND(AVG(satisfaction_rating), 2) AS avg_satisfaction FROM classified_tickets GROUP BY category ORDER BY high_severity_pct DESC;

What is the sentiment breakdown for each original ticket type?

SELECT original_type, sentiment, COUNT(*) AS count, ROUND(100.0 * COUNT(*) / SUM(COUNT(*)) OVER (PARTITION BY original_type), 1) AS pct FROM classified_tickets GROUP BY original_type, sentiment ORDER BY original_type, count DESC;
